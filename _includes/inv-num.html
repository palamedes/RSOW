{%- comment -%} Format an integer with thousands separators. Usage: {% include inv-num.html n=1234567 %} {%- endcomment -%} {%- assign n = include.n | round -%} {%- assign neg = '' -%} {%- if n < 0 -%}{%- assign neg = '-' -%}{%- assign n = 0 | minus: n -%}{%- endif -%} {%- assign digits = n | append: '' | split: '' -%} {%- assign count = digits | size -%} {{- neg -}} {%- for ch in digits -%} {%- assign fromEnd = count | minus: forloop.index0 -%} {%- assign mod = fromEnd | modulo: 3 -%} {%- if mod == 0 and forloop.index0 != 0 -%},{%- endif -%}{{- ch -}} {%- endfor -%}
